// ==================== CLAUDE AI API ====================
async function callClaudeAPI(messages, systemPrompt) {
  const response = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 1000,
      system: systemPrompt,
      messages: messages
    })
  });
  const data = await response.json();
  if (data.error) throw new Error(data.error.message);
  return data.content.map(c => c.text || '').join('');
}

function computeHealthScore() {
  let score = 100;
  const age = profile.age, w = profile.weight, hb = profile.hb;
  const bpParts = (profile.bp||'120/80').split('/');
  const sys = parseInt(bpParts[0])||120, dia = parseInt(bpParts[1])||80;
  const sug = profile.sugar||95;
  const med = (profile.medical||'').toLowerCase();
  if (age < 18 || age > 65) score -= 20;
  if (w < 50) score -= 20;
  if (hb < 12.5) score -= 25; else if (hb < 13.5 && profile.gender==='male') score -= 10;
  if (sys >= 140 || dia >= 90) score -= 20; else if (sys >= 130) score -= 10;
  if (sug >= 126) score -= 20; else if (sug >= 100) score -= 8;
  if (med && med !== 'none' && med !== '') score -= 10;
  return Math.max(0, Math.min(100, score));
}

// ==================== LOCATION ====================
let myLat=null,myLng=null;
function getMyLoc(){
  document.getElementById('myLocText').textContent='Detecting location...';
  if(!navigator.geolocation){toast('Geolocation not supported','error');return;}
  navigator.geolocation.getCurrentPosition(pos=>{
    myLat=pos.coords.latitude;myLng=pos.coords.longitude;
    document.getElementById('myLocText').textContent=`✅ Your location: ${myLat.toFixed(4)}, ${myLng.toFixed(4)}`;
    document.getElementById('locText').textContent=`${myLat.toFixed(4)}, ${myLng.toFixed(4)}`;
    toast('📍 Location detected!','success');
  },err=>{
    document.getElementById('myLocText').textContent='📍 Default: Chennai, Tamil Nadu';
    document.getElementById('locText').textContent='Chennai, Tamil Nadu';
  });
}

// ==================== NAV ====================
function nav(secId,title){
  document.querySelectorAll('.sec').forEach(s=>s.classList.remove('act'));
  document.querySelectorAll('.sb li').forEach(l=>l.classList.remove('act'));
  const sec=document.getElementById(secId);
  if(sec)sec.classList.add('act');
  document.getElementById('pageTitle').textContent=(T[curLang]&&T[curLang]['nav_'+secId.replace('Sec','')])||title.toUpperCase();
  if(secId==='reportSec')refreshReport();
  if(secId==='bankSec'){renderBanks();getMyLoc();}
  if(secId==='notifSec')renderNotifs();
  if(secId==='invSec')renderInv();
  if(secId==='chatSec')renderChatUsers();
  if(secId==='reqSec')renderReqs('all');
  if(secId==='profileSec')loadProfileToForm();
  if(secId==='dashSec')renderDash();
  if(secId==='donorSec')findDonors();
}

function toast(msg,type='default'){
  const t=document.getElementById('toast');t.textContent=msg;
  t.style.background=type==='success'?'var(--green)':type==='error'?'var(--red)':'var(--dark)';
  t.style.transform='translateY(0)';t.style.opacity='1';
  setTimeout(()=>{t.style.transform='translateY(100px)';t.style.opacity='0';},3000);
}

  document.documentElement.style.setProperty('--bg',isDark?'#12122a':'#f2f3f7');
  document.documentElement.style.setProperty('--card',isDark?'#1e1e36':'#ffffff');

  toast(isDark?'🌙 Dark mode activated':'☀️ Light mode activated');
}
