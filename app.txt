// ================= LOGIN =================
function login(){

// ================= NEARBY HOSPITALS =================
function loadNearbyHospitals(){
  const container = document.querySelector("#hospitalData .cards");
  if(!container) return;
  container.innerHTML = "<p>📍 Getting your location...</p>";
  if(!navigator.geolocation){ container.innerHTML="<p>Geolocation not supported.</p>"; return; }
  navigator.geolocation.getCurrentPosition(pos => {
    const lat = pos.coords.latitude;
    const lng = pos.coords.longitude;
    container.innerHTML = `
      <div style="width:100%;border-radius:12px;overflow:hidden">
        <div style="background:#d90429;color:white;padding:12px 16px;display:flex;justify-content:space-between">
          <span>🏥 Hospitals Near You</span>
          <a href="https://www.google.com/maps/search/hospitals/@${lat},${lng},15z" target="_blank"
            style="color:white;font-size:13px;background:rgba(255,255,255,0.2);padding:4px 10px;border-radius:20px;text-decoration:none">
            🗺️ Open Maps
          </a>
        </div>
        <iframe width="100%" height="450" frameborder="0" style="border:0;display:block"
          src="https://www.openstreetmap.org/export/embed.html?bbox=${lng-0.05},${lat-0.05},${lng+0.05},${lat+0.05}&layer=mapnik&marker=${lat},${lng}">
        </iframe>
        <div style="padding:12px;background:white">
          <p style="color:#666;font-size:13px">📍 Your location: <b>${lat.toFixed(4)}, ${lng.toFixed(4)}</b></p>
        </div>
      </div>`;
    fetchNearbyHospitals(lat, lng);
  }, () => {
    container.innerHTML = `
      <iframe width="100%" height="400" frameborder="0" style="border:0;display:block"
        src="https://www.openstreetmap.org/export/embed.html?bbox=80.22,13.06,80.32,13.10&layer=mapnik">
      </iframe>
      <p style="color:#999;padding:10px">⚠️ Location denied. Showing Chennai.
        <a href="#" onclick="loadNearbyHospitals()" style="color:#d90429">Try again</a>
      </p>`;
  });
}

function fetchNearbyHospitals(lat, lng){
  const listEl = document.getElementById("hospitalList");
  if(!listEl) return;
  listEl.innerHTML = "<p style='color:#999'>🔍 Fetching nearby hospitals...</p>";
  const query = `[out:json][timeout:25];(node["amenity"="hospital"](around:5000,${lat},${lng});way["amenity"="hospital"](around:5000,${lat},${lng}););out body;`;
  fetch("https://overpass-api.de/api/interpreter", { method:"POST", body:query })
    .then(r => r.json())
    .then(data => {
      if(!data.elements||data.elements.length===0){ listEl.innerHTML="<p>No hospitals found nearby.</p>"; return; }
      let html = `<h3 style="color:#d90429;margin:12px 0">🏥 ${data.elements.length} Hospitals Found</h3>`;
      data.elements.forEach((el,i) => {
        const name  = el.tags&&el.tags.name ? el.tags.name : "Hospital "+(i+1);
        const phone = el.tags&&el.tags.phone ? el.tags.phone : "";
        const eLat  = el.lat||lat;
        const eLng  = el.lon||lng;
        html += `
          <div style="background:white;border-left:4px solid #d90429;padding:12px;margin:8px 0;border-radius:8px;box-shadow:0 1px 4px rgba(0,0,0,0.1)">
            <b>🏥 ${name}</b>
            ${phone?`<br><small>📞 ${phone}</small>`:""}
            <br>
            <a href="https://www.google.com/maps/dir/${lat},${lng}/${eLat},${eLng}" target="_blank"
              style="color:#d90429;font-size:13px">🗺️ Get Directions</a>
          </div>`;
      });
      listEl.innerHTML = html;
    })
    .catch(() => { listEl.innerHTML="<p style='color:#999'>Could not fetch hospitals. Check internet.</p>"; });
}

// ================= SAVE PROFILE =================
function saveProfile(){
  const user = auth.currentUser;
  if(!user){ alert("Not logged in"); return; }
  const name       = (document.getElementById("editName")||{}).value||"";
  const mobile     = (document.getElementById("editMobile")||{}).value||"";
  const location   = (document.getElementById("editLocation")||{}).value||"";
  const age        = (document.getElementById("editAge")||{}).value||"";
  const editBlood  = document.getElementById("editBlood");
  const bloodGroup = editBlood ? editBlood.value : "";
  if(!name.trim()||!mobile.trim()||!location.trim()){
    alert("Name, Mobile and Location are required."); return;
  }
  const updateData = { name:name.trim(), mobile:mobile.trim(), location:location.trim(), age:age.trim() };
  if(bloodGroup && bloodGroup !== "Select Blood Group" && bloodGroup !== "") updateData.bloodGroup = bloodGroup;
  const btn = document.querySelector("#editProfilePage button");
  if(btn){ btn.innerHTML="⏳ Saving..."; btn.disabled=true; }
  db.collection("users").doc(user.uid).update(updateData)
    .then(() => {
      if(btn){ btn.innerHTML="💾 Save Changes"; btn.disabled=false; }
      alert("✅ Profile Updated Successfully!");
      loadMyProfile();
      showFeature("profileInfoPage");
    })
    .catch(err => {
      if(btn){ btn.innerHTML="💾 Save Changes"; btn.disabled=false; }
      if(err.code==="not-found"){
        db.collection("users").doc(user.uid).set(updateData,{merge:true})
          .then(()=>{ alert("✅ Profile Saved!"); loadMyProfile(); showFeature("profileInfoPage"); })
          .catch(e=>alert("Failed: "+e.message));
      } else { alert("Update Failed: "+err.message); }
    });
}

function sendMessage(){
  const input   = document.getElementById("messageInput");
  const message = input.value.trim();
  if(!message) return;
  const user = auth.currentUser;
  if(!user){ alert("Please login first"); return; }
  if(selectedChatUser){
    const chatRoomId = [user.uid, selectedChatUser.uid].sort().join("_");
    db.collection("privateMessages").doc(chatRoomId).collection("messages").add({
      uid: user.uid, sender: user.email, message,
      time: firebase.firestore.FieldValue.serverTimestamp()
    });
  } else {
    db.collection("messages").add({
      uid: user.uid, sender: user.email, email: user.email, message,
      time: firebase.firestore.FieldValue.serverTimestamp()
    });
  }
  input.value="";
}

  db.collection("users").where("online","==",true).onSnapshot(snapshot => {
    let html=""; let count=0;
    snapshot.forEach(doc => {
      if(user && doc.id===user.uid) return;
      const u = doc.data();

      const hColor = hScore>=80?"#4caf50":hScore>=50?"#ff9800":"#d90429";

        <p>🩸 ${u.bloodGroup||"N/A"}</p>
        <p>📍 ${u.location||"N/A"}</p>
        <p style="color:green">🟢 Online Now</p>
        <p style="color:${hColor};font-weight:bold">❤️ ${hScore}% Health</p>
        <p style="color:${u.canDonate?"#4caf50":"#d90429"};font-weight:bold">

db.collection("alerts").orderBy("time","desc").limit(1).onSnapshot(snapshot => {
  snapshot.docChanges().forEach(change => {
    if(change.type==="added" && soundEnabled){
      try {
        const ctx = new (window.AudioContext||window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        osc.connect(ctx.destination);
        osc.frequency.value=880;
        osc.start(); osc.stop(ctx.currentTime+0.3);
      } catch(e){}
    }
  });
});

// ================= SEND EMERGENCY =================
function sendEmergency(){
  const blood    = document.getElementById("bloodGroup").value;
  const hospital = document.getElementById("hospital").value.trim();
  if(!hospital){ alert("Please enter hospital name"); return; }
  const user = auth.currentUser;
  db.collection("alerts").add({
    blood, hospital,
    sentBy: user?user.email:"anonymous",
    uid:    user?user.uid:"",
    time:   firebase.firestore.FieldValue.serverTimestamp()
  }).then(() => {
    if(user){
      db.collection("donations").add({
        uid: user.uid, bloodGroup: blood, hospital,
        time: firebase.firestore.FieldValue.serverTimestamp()
      });
    }
    alert("🚨 Emergency Alert Sent!");
    document.getElementById("hospital").value="";
  }).catch(err=>alert(err.message));
}

// ================= LOCATION =================
function getLocation(){
  if(!navigator.geolocation){ alert("Geolocation not supported"); return; }
  navigator.geolocation.getCurrentPosition(pos => {
    const lat = pos.coords.latitude.toFixed(4);
    const lng = pos.coords.longitude.toFixed(4);
    const el  = document.getElementById("locationText");
    if(el) el.innerHTML=`📍 Your Location:<br>
      <b>Latitude:</b> ${lat}<br><b>Longitude:</b> ${lng}<br>
      <small><a href="https://maps.google.com/?q=${lat},${lng}" target="_blank">🗺️ View on Google Maps</a></small>`;
  }, () => {
    const el = document.getElementById("locationText");
    if(el) el.innerHTML="📍 Location access denied.";
  });
}

// ================= BLOOD REPORTS =================
function loadBloodReports(){
  const el = document.getElementById("bloodReportContent");
  if(!el) return;
  el.innerHTML="<p style='text-align:center;color:#999'>📊 Loading reports...</p>";
  const groups = ["A+","B+","O+","AB+","A-","B-","O-","AB-"];
  const counts = {};
  let done=0;
  groups.forEach(group => {
    db.collection("users").where("bloodGroup","==",group).get().then(snapshot => {
      counts[group]=snapshot.size; done++;
      if(done===groups.length) renderBloodReport(counts);
    });
  });
}

// ================= BLOOD CHECKUP — LOAD USER DATA =================
function loadBloodCheckup(){
  const user = auth.currentUser;
  if(!user) return;
  checkupScores = {};
  db.collection("users").doc(user.uid).onSnapshot(doc => {
    if(!doc.exists) return;
    const d = doc.data();
    const set = (id,val) => { const el=document.getElementById(id); if(el) el.innerHTML=val||"--"; };
    set("checkupName",  "👤 "+(d.name||"User"));
    set("checkupBlood", "🩸 Blood Group: <b>"+(d.bloodGroup||"N/A")+"</b>");
    set("checkupAge",   "🎂 Age: "+(d.age||"N/A")+" | 📍 "+(d.location||"N/A"));
    const photo = document.getElementById("checkupPhoto");
    if(photo) photo.src = d.image||"https://cdn-icons-png.flaticon.com/512/3135/3135715.png";
    let score=0;
    const age = parseInt(d.age)||0;
    if(age>=18 && age<=65) score+=25;
    if((d.bloodUnits||0)>=2) score+=25;
    if(d.name && d.mobile && d.bloodGroup && d.location) score+=25;
    if(d.bloodGroup && d.bloodGroup!=="") score+=25;
    updateOverallScore("profile", score);
  });
  loadCheckupHistory();
}

// ================= PERCENTAGE BAR =================
function percentBar(pct, color){
  return `
    <div style="background:#eee;border-radius:20px;height:16px;overflow:hidden;margin:8px 0">
      <div style="width:${pct}%;background:${color};height:100%;border-radius:20px;transition:width 1.2s ease"></div>
    </div>
    <div style="text-align:right;font-size:14px;font-weight:bold;color:${color}">${pct}%</div>`;
}

// ================= UPDATE OVERALL SCORE =================
function updateOverallScore(type, pct){
  checkupScores[type] = pct;
  const values  = Object.values(checkupScores);
  const overall = Math.round(values.reduce((a,b)=>a+b,0)/values.length);
  const color   = overall>=80?"#4caf50":overall>=50?"#ff9800":"#d90429";
  const circleEl = document.getElementById("overallScorePct");
  if(circleEl) circleEl.innerHTML = overall+"%";
  const circle = document.getElementById("overallScoreCircle");
  if(circle) circle.style.borderColor = color;
  const statusEl = document.getElementById("myHealthStatus");
  if(statusEl) statusEl.innerHTML = `<span style="color:${color};font-weight:bold">${overall}% Health</span>`;
  const badge = document.getElementById("donationBadge");
  if(badge){
    if(overall>=80){ badge.innerHTML="✅ "+overall+"% — YOU ARE ELIGIBLE TO DONATE BLOOD! 🩸"; badge.style.background="rgba(76,175,80,0.3)"; }
    else if(overall>=50){ badge.innerHTML="⚠️ "+overall+"% — PARTIALLY ELIGIBLE. Improve health."; badge.style.background="rgba(255,152,0,0.3)"; }
    else { badge.innerHTML="❌ "+overall+"% — NOT ELIGIBLE. Complete all checkup tabs."; badge.style.background="rgba(217,4,41,0.3)"; }
  }
  const user = auth.currentUser;
  if(user){
    db.collection("users").doc(user.uid).update({
      healthScore: overall,
      canDonate:   overall>=80,
      lastCheckup: new Date()
    });
  }
}

// ================= CHECK HEMOGLOBIN =================
function checkHemoglobin(){
  const val    = parseFloat(document.getElementById("hbValue").value);
  const gender = document.getElementById("hbGender").value;
  const el     = document.getElementById("hbResult");
  if(!val){ el.innerHTML="<p style='color:red'>Please enter hemoglobin value.</p>"; return; }
  let min,max,label,color,advice;
  if(gender==="male")          { min=13.5; max=17.5; }
  else if(gender==="female")   { min=12.0; max=15.5; }
  else if(gender==="pregnant") { min=11.0; max=14.0; }
  else                         { min=11.5; max=15.5; }
  const pct = Math.min(100,Math.round((val/max)*100));
  if(val<min){      label="LOW — Anemia Detected 🔴";        color="#d90429"; advice="Eat iron-rich foods. NOT eligible to donate now."; }
  else if(val>max){ label="HIGH — Polycythemia Possible 🟡"; color="#ff9800"; advice="Consult a doctor. Stay hydrated."; }
  else {            label="NORMAL — Healthy Level ✅";        color="#4caf50"; advice="Hemoglobin healthy! Likely eligible to donate!"; }
  saveCheckupRecord({ type:"Hemoglobin", result:label, value:val, gender });
  updateOverallScore("hb", pct);
  el.innerHTML=`
    <div style="border-left:4px solid ${color};background:${color}11;border-radius:10px;padding:20px">
      <h4 style="color:${color}">${label}</h4>
      <p>Value: <b style="font-size:22px;color:${color}">${val} g/dL</b></p>
      <p style="color:#666">Normal: ${min}–${max} g/dL</p>
      ${percentBar(pct,color)}
      <div style="background:white;border-radius:8px;padding:12px;margin-top:12px"><b>💡</b> ${advice}</div>
      <p style="color:#999;font-size:12px;margin-top:10px">⚠️ Not a medical diagnosis.</p>
    </div>`;
}

// ================= CHECK BLOOD PRESSURE =================
function checkBloodPressure(){
  const sys = parseInt(document.getElementById("bpSystolic").value);
  const dia = parseInt(document.getElementById("bpDiastolic").value);
  const el  = document.getElementById("bpResult");
  if(!sys||!dia){ el.innerHTML="<p style='color:red'>Please enter both values.</p>"; return; }

  if(sys<90||dia<60){        label="LOW BP 🔵";       color="#2196f3"; healthPct=60; advice="Drink water. Eat salty foods."; }

  const el   = document.getElementById("sugarResult");
  if(!val){ el.innerHTML="<p style='color:red'>Please enter sugar level.</p>"; return; }
  let label,color,advice,pct;
  if(type==="fasting"){
    if(val<70){        label="LOW SUGAR 🔵";    color="#2196f3"; pct=50;  advice="Eat something sweet immediately."; }
    else if(val<=100){ label="NORMAL ✅";         color="#4caf50"; pct=100; advice="Fasting sugar normal. Eligible!"; }
    else if(val<=125){ label="PRE-DIABETES 🟡"; color="#ff9800"; pct=55;  advice="Reduce sugar. Consult doctor."; }
    else {             label="DIABETES 🔴";      color="#d90429"; pct=20;  advice="Consult doctor. NOT eligible."; }
  } else {
    if(val<70){        label="LOW SUGAR 🔵";    color="#2196f3"; pct=50;  advice="Eat something sweet immediately."; }
    else if(val<140){  label="NORMAL ✅";         color="#4caf50"; pct=100; advice="Post-meal sugar normal. Eligible!"; }
    else if(val<200){  label="PRE-DIABETES 🟡"; color="#ff9800"; pct=55;  advice="Reduce carbs. Exercise after meals."; }
    else {             label="DIABETES 🔴";      color="#d90429"; pct=20;  advice="High sugar. NOT eligible."; }
  }
  saveCheckupRecord({ type:"Blood Sugar", result:label, value:val, testType:type });
  updateOverallScore("sugar", pct);
  el.innerHTML=`
    <div style="border-left:4px solid ${color};background:${color}11;border-radius:10px;padding:20px">
      <h4 style="color:${color}">${label}</h4>
      <p>Value: <b style="font-size:22px;color:${color}">${val} mg/dL</b> (${type==="fasting"?"Fasting":"After Meal"})</p>
      ${percentBar(pct,color)}
      <div style="background:white;border-radius:8px;padding:12px;margin-top:12px"><b>💡</b> ${advice}</div>
      <p style="color:#999;font-size:12px;margin-top:10px">⚠️ Not a medical diagnosis.</p>
    </div>`;
}

// ================= CHECK ELIGIBILITY =================
function checkEligibility(){
  const questions=["eli_age","eli_weight","eli_hb","eli_bp","eli_healthy","eli_notsick","eli_notatoo","eli_nosurg"];
  let yesCount=0, unanswered=0;
  questions.forEach(q=>{
    const ans=document.querySelector(`input[name="${q}"]:checked`);
    if(!ans) unanswered++;
    else if(ans.value==="yes") yesCount++;
  });
  const el=document.getElementById("eligibilityResult");
  if(unanswered>0){ el.innerHTML=`<p style="color:#ff9800">⚠️ Please answer all ${unanswered} remaining questions.</p>`; return; }
  const pct   = Math.round((yesCount/questions.length)*100);
  const color = pct===100?"#4caf50":pct>=75?"#ff9800":"#d90429";
  let label,advice;
  if(pct===100){ label="✅ FULLY ELIGIBLE TO DONATE!"; advice="You meet ALL criteria. Go donate and save a life! 🩸"; }
  else if(pct>=75){ label="⚠️ PARTIALLY ELIGIBLE"; advice="Consult a doctor before donating."; }
  else { label="❌ NOT ELIGIBLE NOW"; advice="Focus on health. Try again later."; }
  saveCheckupRecord({ type:"Eligibility", result:label, score:yesCount, total:questions.length });
  updateOverallScore("eligibility", pct);
  el.innerHTML=`
    <div style="border-left:4px solid ${color};background:${color}11;border-radius:10px;padding:20px">
      <h4 style="color:${color}">${label}</h4>
      <p>Score: <b style="font-size:22px;color:${color}">${yesCount}/${questions.length}</b></p>
      ${percentBar(pct,color)}
      <div style="background:white;border-radius:8px;padding:12px;margin-top:12px"><b>💡</b> ${advice}</div>
    </div>`;
}

// ================= SAVE CHECKUP RECORD =================
function saveCheckupRecord(data){
  const user=auth.currentUser;
  if(!user) return;
  db.collection("checkups").add({ uid:user.uid, ...data, time:firebase.firestore.FieldValue.serverTimestamp() });
}

// ================= LOAD CHECKUP HISTORY =================
function loadCheckupHistory(){
  const user=auth.currentUser;
  if(!user) return;
  const el=document.getElementById("checkupHistoryList");
  if(!el) return;
  el.innerHTML="<p style='color:#999'>Loading...</p>";
  db.collection("checkups").where("uid","==",user.uid).orderBy("time","desc").limit(20)
    .onSnapshot(snapshot=>{
      if(snapshot.empty){ el.innerHTML="<p style='color:#999'>No checkup history yet.</p>"; return; }
      let html="";
      snapshot.forEach(doc=>{
        const d=doc.data();
        const t=d.time?new Date(d.time.toDate()).toLocaleString():"Just now";
        const color=d.result&&(d.result.includes("NORMAL")||d.result.includes("ELIGIBLE"))?"#4caf50":
                    d.result&&d.result.includes("LOW")?"#2196f3":
                    d.result&&(d.result.includes("HIGH")||d.result.includes("Anemia"))?"#d90429":"#ff9800";
        html+=`<div style="background:white;border-left:4px solid ${color};padding:12px;margin:8px 0;border-radius:8px;box-shadow:0 1px 4px rgba(0,0,0,0.08)">
          <div style="display:flex;justify-content:space-between">
            <b style="color:${color}">${d.type||"Checkup"}</b>
            <small style="color:#999">${t}</small>
          </div>
          <p style="margin-top:4px;color:#333">${d.result||""}</p>
        </div>`;
      });
      el.innerHTML=html;
    });
}

// ================= OPEN NEAREST HOSPITAL =================
function openNearestHospital(){
  if(!navigator.geolocation){ alert("Geolocation not supported"); return; }
  navigator.geolocation.getCurrentPosition(pos => {
    const lat = pos.coords.latitude;
    const lng = pos.coords.longitude;
    window.open("https://www.google.com/maps/search/hospitals/@"+lat+","+lng+",15z","_blank");
  }, () => {
    window.open("https://www.google.com/maps/search/hospitals","_blank");
  });
}

function logout(){
  const user=auth.currentUser;
  const doLogout=()=>auth.signOut().then(()=>window.location.href="login.html");
  if(user){ db.collection("users").doc(user.uid).update({online:false}).then(doLogout).catch(doLogout); }
  else doLogout();
}
